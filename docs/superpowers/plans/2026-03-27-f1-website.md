# F1 信息网站 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a multi-page F1 information website using vanilla HTML/CSS/JS, connected to OpenF1 and Jolpica APIs, with desktop split-panel and mobile full-page navigation.

**Architecture:** Multi-page HTML files share a CSS design system and a JS module layer. The API layer (`js/api/`) is pure data with zero DOM dependency, making it reusable for WeChat Mini Program migration. Pages use ES modules (`<script type="module">`) served via the existing localhost:3000 static server.

**Tech Stack:** Vanilla HTML5, CSS custom properties, ES Modules (no bundler), OpenF1 API, Jolpica API, Canvas (charts), `Intl.DateTimeFormat` (timezone), `fetch()` (requests)

```js
const BASE = 'https://api.openf1.org/v1'

async function get(path, params = {}) {
  const url = new URL(BASE + path)
  Object.entries(params).forEach(([k, v]) => {
    if (v != null) url.searchParams.set(k, v)
  })
  const res = await fetch(url)
  if (!res.ok) throw new Error(`OpenF1 ${res.status} ${path}`)
  return res.json()
}
```

```js
// Format ISO UTC string to user's local time
export function formatLocalTime(isoUtc, options = {}) {
  if (!isoUtc) return '—'
  const tz = Intl.DateTimeFormat().resolvedOptions().timeZone
  return new Date(isoUtc).toLocaleString('default', { timeZone: tz, ...options })
}

// Format date only: "Mar 27"
export function formatDate(isoUtc) {
  if (!isoUtc) return '—'
  return formatLocalTime(isoUtc, { month: 'short', day: 'numeric' })
}

// Format time only: "14:00"
export function formatTime(isoUtc) {
  if (!isoUtc) return '—'
  return formatLocalTime(isoUtc, { hour: '2-digit', minute: '2-digit' })
}

// Countdown: "3d 4h" or "45m" or "Started"
export function formatCountdown(isoUtc) {
  if (!isoUtc) return '—'
  const diff = new Date(isoUtc) - Date.now()
  if (diff <= 0) return 'Started'
  const d = Math.floor(diff / 86400000)
  const h = Math.floor((diff % 86400000) / 3600000)
  const m = Math.floor((diff % 3600000) / 60000)
  if (d > 0) return `${d}d ${h}h`
  if (h > 0) return `${h}h ${m}m`
  return `${m}m`
}

// Country code → flag emoji: "AU" → 🇦🇺
export function countryFlag(code) {
  if (!code || code.length !== 2) return ''
  return [...code.toUpperCase()].map(c =>
    String.fromCodePoint(0x1F1E6 + c.charCodeAt(0) - 65)
  ).join('')
}

// Seconds → lap time: 83.456 → "1:23.456"
export function formatLapTime(seconds) {
  if (seconds == null || isNaN(seconds)) return '—'
  const m = Math.floor(seconds / 60)
  const s = (seconds % 60).toFixed(3).padStart(6, '0')
  return `${m}:${s}`
}

// Animate number from 0 to target
export function animateNumber(el, target, duration = 500) {
  const start = performance.now()
  const n = parseFloat(target)
  if (isNaN(n)) { el.textContent = target; return }
  function step(now) {
    const t = Math.min((now - start) / duration, 1)
    const ease = 1 - Math.pow(1 - t, 3) // ease-out-cubic
    el.textContent = Math.round(n * ease)
    if (t < 1) requestAnimationFrame(step)
  }
  requestAnimationFrame(step)
}
```

```js
import { openF1 } from './api/openf1.js'

export const YEAR = new Date().getFullYear()

let pollingTimer = null

// Detect if we're in a race weekend (±3 days of any session)
export async function detectRaceState() {
  const sessions = await openF1.getSessions({ year: YEAR })
  const now = Date.now()
  const window_ms = 3 * 24 * 60 * 60 * 1000 // 3 days

  const nearby = sessions.filter(s => {
    const start = new Date(s.date_start).getTime()
    return Math.abs(start - now) < window_ms
  })

  if (nearby.length === 0) {
    return { state: 'NO_RACE', year: YEAR, sessions }
  }

  const meetingKey = nearby[0].meeting_key
  const meetingSessions = sessions.filter(s => s.meeting_key === meetingKey)

  // Find the currently active session (started but not yet +3h past end)
  const currentSession = meetingSessions.find(s => {
    const start = new Date(s.date_start).getTime()
    const end = s.date_end
      ? new Date(s.date_end).getTime()
      : start + 3 * 60 * 60 * 1000
    return now >= start && now <= end + 30 * 60 * 1000 // 30min grace
  }) || null

  // Find next upcoming session
  const nextSession = meetingSessions
    .filter(s => new Date(s.date_start).getTime() > now)
    .sort((a, b) => new Date(a.date_start) - new Date(b.date_start))[0] || null

  // Most recently completed session
  const lastSession = meetingSessions
    .filter(s => new Date(s.date_start).getTime() < now)
    .sort((a, b) => new Date(b.date_start) - new Date(a.date_start))[0] || null

  return {
    state: 'RACE_WEEKEND',
    year: YEAR,
    meetingKey,
    sessions: meetingSessions,
    currentSession,
    nextSession,
    lastSession,
    allSessions: sessions,
  }
}

export function startPolling(callback, intervalMs = 30_000) {
  stopPolling()
  callback() // immediate first call
  pollingTimer = setInterval(callback, intervalMs)
}

export function stopPolling() {
  if (pollingTimer) { clearInterval(pollingTimer); pollingTimer = null }
}
```

```js
// Desktop split-panel: fetch a detail page's #app content and show in panel
// On mobile (< 1024px), navigate directly instead.

const BREAKPOINT = 1024

let currentHref = null

export function initPanel(listEl, panelEl) {
  // Wire all .standing-row and .card links inside listEl
  listEl.addEventListener('click', e => {
    const link = e.target.closest('a[href]')
    if (!link) return
    const href = link.getAttribute('href')
    if (!href || href.startsWith('http') || href.startsWith('#')) return

    if (window.innerWidth < BREAKPOINT) return // let browser navigate normally

    e.preventDefault()

    // Mark active
    listEl.querySelectorAll('.active').forEach(el => el.classList.remove('active'))
    link.classList.add('active')

    loadPanel(href, panelEl)
  })
}

async function loadPanel(href, panelEl) {
  if (href === currentHref) return
  currentHref = href

  panelEl.classList.remove('hidden')

  // Show skeleton while loading
  panelEl.innerHTML = `<div class="split-panel-inner">
    <div class="skeleton skel-title" style="margin-bottom:16px"></div>
    <div class="skeleton skel-text"></div>
    <div class="skeleton skel-text-sm"></div>
    <div style="margin-top:16px"></div>
    <div class="skeleton skel-row"></div>
    <div class="skeleton skel-row"></div>
    <div class="skeleton skel-row"></div>
  </div>`

  try {
    const res = await fetch(href)
    const text = await res.text()
    const doc = new DOMParser().parseFromString(text, 'text/html')
    const content = doc.getElementById('app')

    // Re-run the page's module script to populate #app
    // Since we can't re-run modules, we embed the rendered HTML
    // The detail page must expose a renderPage() export we can call
    // Instead: use a URL param to signal panel-mode; page renders and we grab it

    // Simpler approach: fetch the page's data directly in this session
    // We delegate to a panel-render function registered by each detail page
    const renderer = panelRenderers[href.split('?')[0]]
    if (renderer) {
      const params = new URLSearchParams(href.split('?')[1] || '')
      panelEl.innerHTML = `<div class="split-panel-inner"></div>`
      const inner = panelEl.querySelector('.split-panel-inner')
      await renderer(params, inner)
    } else {
      panelEl.innerHTML = `<div class="split-panel-inner">${content?.innerHTML || '<p class="text-2">无法加载</p>'}</div>`
    }
  } catch (err) {
    panelEl.innerHTML = `<div class="split-panel-inner"><p class="text-2">加载失败</p></div>`
  }
}

// Registry: detail pages register a renderer for panel mode
// Usage: registerPanelRenderer('driver.html', async (params, el) => { ... })
const panelRenderers = {}

export function registerPanelRenderer(page, fn) {
  panelRenderers[page] = fn
}

export function closePanel(panelEl) {
  panelEl.classList.add('hidden')
  currentHref = null
}
```

```js
    s.Driver.driverId === driverCode.toLowerCase()
  )

  if (!standing) {
    app.innerHTML = `<p class="text-2">找不到车手：${driverCode}</p>`
    return
  }

  const driver = standing.Driver
  const constructor = standing.Constructors?.[0]
  const teamInfo = getTeamInfo(constructor?.name)
  const teamColor = teamInfo?.color || '#555'
  const flag = countryFlag(driver.nationality?.slice(0, 2) || '')

  // Get all race results for this driver this season
  const resultsData = await fetch(
    `https://api.jolpi.ca/ergast/f1/${YEAR}/drivers/${driver.driverId}/results.json?limit=30`
  ).then(r => r.json()).catch(() => null)
  const races = resultsData?.MRData?.RaceTable?.Races || []

  // Calc stats from results
  const wins = races.filter(r => r.Results?.[0]?.position === '1').length
  const poles = races.filter(r => r.Results?.[0]?.grid === '1').length
  const podiums = races.filter(r => parseInt(r.Results?.[0]?.position) <= 3).length
  const fastestLaps = races.filter(r => r.Results?.[0]?.FastestLap?.rank === '1').length

  // Get headshot from OpenF1 (use latest session)
  const allSessions = await openF1.getSessions({ year: YEAR })
  const raceSession = allSessions.filter(s => s.session_name === 'Race')
    .sort((a, b) => new Date(b.date_start) - new Date(a.date_start))[0]
  let headshotUrl = ''
  if (raceSession) {
    const drivers = await openF1.getDrivers(raceSession.session_key)
    const d = drivers.find(d => d.name_acronym === driverCode.toUpperCase())
    headshotUrl = d?.headshot_url || ''
  }

  app.innerHTML = `
    ${breadcrumb([{ label: '车手', href: 'drivers.html' }, { label: driver.familyName }])}

    <div class="hero-strip fade-up" style="border-left: 4px solid ${teamColor}; margin-bottom:28px">
      <div style="display:flex;align-items:center;gap:16px;flex:1">
        ${headshotUrl
          ? `<img src="${headshotUrl}" alt="${driver.givenName}" style="height:80px;width:auto;border-radius:8px;object-fit:cover">`
          : `<div style="width:60px;height:80px;background:var(--color-surface-2);border-radius:8px;border:1px solid var(--color-border)"></div>`
        }
        <div>
          <div class="label" style="color:${teamColor};margin-bottom:4px">${constructor?.name || ''}</div>
          <div class="page-title">${driver.givenName} ${driver.familyName.toUpperCase()}</div>
          <div class="card-sub" style="margin-top:4px">
            #${driver.permanentNumber} · ${flag} ${driver.nationality}
          </div>
        </div>
      </div>
      <div class="hero-strip-side">
        <div style="font-size:48px;font-weight:900;color:${teamColor};font-family:var(--font-mono);line-height:1">#${driver.permanentNumber}</div>
        <div class="countdown-label">Driver No.</div>
      </div>
    </div>

    ${statGrid([
      { value: standing.points, label: '积分', animate: true },
      { value: standing.position, label: '排名', animate: true },
      { value: wins, label: '胜场', animate: true },
      { value: podiums, label: '上台', animate: true },
      { value: poles, label: '杆位', animate: true },
      { value: fastestLaps, label: '最快圈', animate: true },
    ])}

    <div class="section-title" style="margin-bottom:16px">本赛季各站成绩</div>
    <table class="result-table">
      <thead><tr>
        <th>站次</th><th>比赛</th><th>排位</th><th>正赛</th><th class="mono-cell">积分</th>
      </tr></thead>
      <tbody>
        ${races.map(r => {
          const res = r.Results?.[0]
          return `<tr>
            <td class="mono-cell text-2">R${r.round}</td>
            <td><a href="race.html?meeting_key=${r.round}" style="color:inherit">${r.raceName.replace(' Grand Prix','')}</a></td>
            <td class="mono-cell">${res?.grid || '—'}</td>
            <td class="mono-cell">${res?.position || res?.status || '—'}</td>
            <td class="mono-cell">${res?.points || '0'}</td>
          </tr>`
        }).join('')}
      </tbody>
    </table>
  `

  triggerStatAnimations(app)
}

init().catch(err => {
  app.innerHTML = `<p class="text-2">车手数据加载失败：${err.message}</p>`
})
```

```js
        d.team_name === standing?.Constructor?.name ||
```
